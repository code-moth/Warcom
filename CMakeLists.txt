cmake_minimum_required(VERSION 3.16)
project(warcom LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(warcom-engine core/main.cpp core/resolver.cpp)
add_executable(warcom-reference core/main.cpp core/reference.cpp)
enable_testing()
find_program(NODE_EXECUTABLE node REQUIRED)
add_test(NAME core-regression COMMAND ${NODE_EXECUTABLE} --test ${CMAKE_SOURCE_DIR}/tests/core.test.mjs)
set_tests_properties(core-regression PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} ENVIRONMENT "WARCOM_ENGINE=$<TARGET_FILE:warcom-engine>;WARCOM_REFERENCE=$<TARGET_FILE:warcom-reference>")
